Before · coupled 
async function cancelOrder(db: Db, id: number) {
  const row = db.get("SELECT status FROM orders WHERE id = ?", id);
  if (row.status === "cancelled") throw new Error("already cancelled");
  db.run("UPDATE orders SET status = 'cancelled' WHERE id = ?", id);
}
After · port + adapter 
class CancelOrder {
  constructor(private readonly orders: OrderRepository) {}
  execute(id: number) {
    const order = this.orders.get(id) ?? throwNotFound(id);
    order.cancel();
    this.orders.save(order);
  }
}
